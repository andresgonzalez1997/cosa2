SELECT
    material_number AS Product_ID,
    fiscal_year AS FY,
    posting_period AS Month_,
    posting_date,

    sales_order_number,
    sales_order_item,

    additional_quantity_in_tons AS Tons,

    amount_usd,

    cfmr_account_level1,
    cfmr_account_level2,
    cfmr_account_level3,
    cfmr_account_level4,
    cfmr_account_level5,
    cfmr_account_description

FROM prd_product_anh_external_datablocks.tca_enhanced_profit_and_loss

WHERE UPPER(sales_org) = 'HN10'

  AND UPPER(TRIM(cfmr_account_level2))
      = 'COST OF INVENTORY SOLD'

  AND UPPER(TRIM(cfmr_account_level3))
      = 'MATERIAL COSTS'

  AND amount_usd <> 0

ORDER BY
    posting_date DESC,
    material_number

LIMIT 50;
