WITH tm_real_documento AS (
    -- TM sin restringir a Notas de Credito/Debito, evita heredar tonelada arbitraria
    SELECT
        billing_document_number,
        SUM(additional_quantity_in_tons) AS tm_real
    FROM prd_product_anh_external_datablocks.tca_enhanced_profit_and_loss
    WHERE billing_document_number IS NOT NULL
    GROUP BY billing_document_number
),

material_uom AS (
    -- Una sola UM por producto.
    -- Validado previamente: no existen sap_product_id con mas de una
    -- material_base_unit_of_measure distinta.
    SELECT
        sap_product_id,
        MAX(material_base_unit_of_measure) AS material_base_unit_of_measure
    FROM prd_product_anh_internal_datablocks.tca_movements
    WHERE sap_product_id IS NOT NULL
      AND material_base_unit_of_measure IS NOT NULL
    GROUP BY sap_product_id
)

SELECT
    -- ============================================================
    -- CAMPOS ORIGINALES
    -- ============================================================

    profit.company_code_currency AS "Tipo_Moneda",
    profit.amount_usd AS "Moneda",
    profit.sales_order_number AS "Orden de venta",
    profit.billing_document_number AS "billing_document_number",
    profit.amount_company_code_currency AS "Moneda Local",
    profit.fiscal_year AS "Fiscal Year",
    profit.posting_period AS "Month Desc",
    profit.material_number AS "product id",
    product.form_code AS "form",
    net_contents.net_contents AS "Container",
    product.unit_type_code AS "empaque",
    org.country_name AS "Country",
    COALESCE(profit.profit_center, profit.cost_center) AS "Center",
    customer.customer_segment_name AS "Customer Segment",
    profit.source_customer_sold_to_name AS "Customer Name",
    profit.account_number AS "Cuenta Contable",
    profit.amount_usd AS "Amount (CY & LY)",
    profit.amount_company_code_currency AS "Amount (CY & LY) Local",
    profit.posting_date AS "Date",
    profit.account_description AS "OHONO1",
    profit.customer_sold_to_number AS "Customer Number",
    profit.material_number AS "ID_MatCod",

    -- TM crudo (volumen total) para ratios como Discount/TM
    profit.additional_quantity_in_tons AS "TM",

    -- TM contado UNA sola vez por linea
    -- Evita duplicar tonelada por cada cuenta del P&L
    CASE
        WHEN ROW_NUMBER() OVER (
            PARTITION BY
                profit.sales_order_number,
                profit.sales_order_item,
                profit.product_id,
                profit.material_number,
                profit.org_id,
                profit.customer_shipto_id,
                profit.fiscal_year,
                profit.posting_period,
                profit.posting_date
            ORDER BY
                profit.accounting_document_number DESC,
                profit.ledger_posting_item DESC
        ) = 1
        THEN profit.additional_quantity_in_tons
    END AS "TM Unique",

    -- TM solo de lineas de descuento
    -- Segun mapeo de negocio: condicion DISCOUNT
    CASE
        WHEN profit.cfmr_account_description ILIKE '%DISCOUNT%'
            THEN profit.additional_quantity_in_tons
    END AS "TM Discount",

    CASE
        WHEN profit.cfmr_account_description ILIKE '%DISCOUNT%'
            THEN profit.account_description
    END AS "Discount Description",

    profit.sap_material_description AS "Material Name",
    profit.account_number AS "No_Desc",

    CASE
        WHEN profit.cfmr_account_description ILIKE '%DISCOUNT%'
            THEN profit.amount_usd
    END AS "Discount Amount",

    CASE
        WHEN profit.cfmr_account_description ILIKE '%DISCOUNT%'
            THEN profit.amount_company_code_currency
    END AS "Discount Amount Local",

    CASE
        WHEN profit.cfmr_account_description ILIKE '%DISCOUNT%'
            THEN profit.account_number
    END AS "ID Discount",

    CASE
        WHEN profit.cfmr_account_description ILIKE '%DISCOUNT%'
            THEN profit.account_description
    END AS "Discount Name",

    weekofyear(profit.posting_date) AS "week",

    CASE
        WHEN profit.cfmr_account_description ILIKE '%DISCOUNT%'
            THEN profit.amount_usd
    END AS "Discount Value",

    CASE
        WHEN profit.cfmr_account_description ILIKE '%DISCOUNT%'
            THEN profit.amount_company_code_currency
    END AS "Discount Value Local",

    profit.sap_material_description AS "Material Description",
    profit.material_number AS "ID Mat",
    product.species_l2 AS "material_category",
    profit.billing_type_code AS "Billing Type Code",
    profit.document_type AS "Document Type",

    -- ============================================================
    -- CAMPOS QUE YA EXISTIAN PARA EL MODELO RELACIONAL
    -- ============================================================

    profit.company_code AS "d_country",
    org.country_name AS "d_descript",
    customer.customer_soldto_id AS "customer_id",
    customer.soldto_display_name AS "customer_name",
    product.material_id AS "material_id",
    product.material_name AS "material_name",

    -- Valor del descuento: SOLO lineas de descuento
    CASE
        WHEN profit.cfmr_account_description ILIKE '%DISCOUNT%'
            THEN profit.amount_usd
    END AS "discount_value",

    CASE
        WHEN profit.cfmr_account_description ILIKE '%DISCOUNT%'
            THEN profit.amount_company_code_currency
    END AS "discount_value_local",

    profit.posting_date AS "discount_date",

    -- Campo libre / manual equivalente a SGTXT en SAP
    profit.item_text AS "Descripcion_Libre",

    -- TM calculado a nivel documento
    tm_real_documento.tm_real AS "TM Real",

    -- ============================================================
    -- NUEVAS COLUMNAS SOLICITADAS Y CONFIRMADAS
    -- ============================================================

    -- Organizacion de ventas
    profit.sales_org AS "Organizaciones",

    -- Cliente SAP
    profit.customer_sold_to_number AS "Cliente",

    -- Nombre cliente
    profit.source_customer_sold_to_name AS "Nombre del cliente",

    -- Material SAP
    profit.material_number AS "Material",

    -- Descripcion del material
    profit.sap_material_description AS "Descripcion Material",

    -- Brand/Subbrand desde product_core
    product.brand_code AS "Brand",
    product.sub_brand_code AS "Subbrand",

    -- Condicion de pago disponible directamente en P&L
    profit.payment_terms AS "Cond.pago",

    -- Base Unit of Measure
    material_uom.material_base_unit_of_measure AS "UM"

FROM prd_product_anh_external_datablocks.tca_enhanced_profit_and_loss profit

-- ================================================================
-- ORGANIZACION / PAIS
-- ================================================================
LEFT JOIN prd_product_anh_core_data.org_location_core org
    ON profit.org_id = org.org_id

-- ================================================================
-- CUSTOMER
-- ================================================================
LEFT JOIN prd_product_anh_core_data.customer_core customer
    ON profit.customer_shipto_id = customer.customer_shipto_id

-- ================================================================
-- PRODUCT
-- ================================================================
LEFT JOIN prd_product_anh_core_data.product_core product
    ON profit.product_id = product.product_id

-- ================================================================
-- NET CONTENTS
-- Se reduce primero a una fila por material para evitar multiplicar
-- registros del P&L.
-- ================================================================
LEFT JOIN (
    SELECT
        material_number,
        MAX(characteristic_value) AS net_contents
    FROM prd_product_anh_internal_datablocks.tca_material_characteristics_assignments
    WHERE characteristic_description = 'NET CONTENTS'
    GROUP BY material_number
) net_contents
    ON profit.material_number = net_contents.material_number

-- ================================================================
-- TM REAL
-- ================================================================
LEFT JOIN tm_real_documento
    ON profit.billing_document_number = tm_real_documento.billing_document_number

-- ================================================================
-- BASE UNIT OF MEASURE
-- tca_movements tiene multiples movimientos por material.
-- material_uom ya la redujo a una sola fila por sap_product_id.
-- ================================================================
LEFT JOIN material_uom
    ON profit.material_number = material_uom.sap_product_id

WHERE org.country_name IN (
        'Guatemala',
        'Honduras',
        'Nicaragua',
        'El Salvador',
        'Costa Rica',
        'Panama',
        'Aruba',
        'Republica Dominicana',
        'Puerto Rico'
    )

    -- Requerimiento de negocio:
    -- solo Regalias / Notas de Credito / Notas de Debito.
    -- No facturas normales.
    AND profit.billing_type_code IN (
        'ZUFD',
        'RE',
        'S1',
        'L2',
        'G2'
    );
