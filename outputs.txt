SELECT DISTINCT
    cfmr_account_level1,
    cfmr_account_level2,
    cfmr_account_level3,
    cfmr_account_level4,
    cfmr_account_level5,
    cfmr_account_description
FROM prd_product_anh_external_datablocks.tca_enhanced_profit_and_loss
WHERE UPPER(sales_org) = 'HN10'
  AND material_type = 'ZMAT'
  AND (
        UPPER(cfmr_account_level1) LIKE '%COST%'
     OR UPPER(cfmr_account_level2) LIKE '%COST%'
     OR UPPER(cfmr_account_level3) LIKE '%COST%'
     OR UPPER(cfmr_account_level4) LIKE '%COST%'
     OR UPPER(cfmr_account_level5) LIKE '%COST%'
     OR UPPER(cfmr_account_description) LIKE '%COST%'
     OR UPPER(cfmr_account_level1) LIKE '%INGREDIENT%'
     OR UPPER(cfmr_account_level2) LIKE '%INGREDIENT%'
     OR UPPER(cfmr_account_level3) LIKE '%INGREDIENT%'
     OR UPPER(cfmr_account_level4) LIKE '%INGREDIENT%'
     OR UPPER(cfmr_account_level5) LIKE '%INGREDIENT%'
     OR UPPER(cfmr_account_description) LIKE '%INGREDIENT%'
     OR UPPER(cfmr_account_level1) LIKE '%PACKAG%'
     OR UPPER(cfmr_account_level2) LIKE '%PACKAG%'
     OR UPPER(cfmr_account_level3) LIKE '%PACKAG%'
     OR UPPER(cfmr_account_level4) LIKE '%PACKAG%'
     OR UPPER(cfmr_account_level5) LIKE '%PACKAG%'
     OR UPPER(cfmr_account_description) LIKE '%PACKAG%'
  )
ORDER BY
    cfmr_account_level1,
    cfmr_account_level2,
    cfmr_account_level3,
    cfmr_account_level4,
    cfmr_account_level5;
