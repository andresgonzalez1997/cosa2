SELECT
    billing_document_number,

    COUNT(*) AS rows_pnl,

    COUNT(DISTINCT sales_order_item) AS sales_order_items,

    COUNT(DISTINCT material_number) AS materials,

    COUNT(DISTINCT account_number) AS accounts,

    SUM(amount_usd) AS total_amount_usd,

    SUM(amount_company_code_currency) AS total_amount_local

FROM prd_product_anh_external_datablocks.tca_enhanced_profit_and_loss

WHERE billing_document_number = 'XXXXXXXXXX'

GROUP BY billing_document_number;
