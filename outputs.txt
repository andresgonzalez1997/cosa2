SELECT
    billing_document_number,

    SUM(
        CASE
            WHEN cfmr_account_description ILIKE '%DISCOUNT%'
            THEN amount_company_code_currency
            ELSE 0
        END
    ) AS discount_local,

    SUM(
        CASE
            WHEN cfmr_account_description NOT ILIKE '%DISCOUNT%'
                 OR cfmr_account_description IS NULL
            THEN amount_company_code_currency
            ELSE 0
        END
    ) AS amount_before_discount_local,

    SUM(amount_company_code_currency) AS total_local,

    COUNT(
        CASE
            WHEN cfmr_account_description ILIKE '%DISCOUNT%'
            THEN 1
        END
    ) AS discount_rows

FROM prd_product_anh_external_datablocks.tca_enhanced_profit_and_loss

WHERE billing_document_number = '9520212283'

GROUP BY billing_document_number;
