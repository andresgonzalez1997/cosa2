SELECT
    billing_document_number,
    sales_order_item,
    material_number,

    COUNT(*) AS rows_item,

    COUNT(additional_quantity_in_tons) AS rows_with_tm,

    COUNT(DISTINCT additional_quantity_in_tons) AS distinct_tm_values,

    SUM(additional_quantity_in_tons) AS sum_tm,

    MAX(additional_quantity_in_tons) AS max_tm

FROM prd_product_anh_external_datablocks.tca_enhanced_profit_and_loss

WHERE billing_type_code IN (
    'ZUFD',
    'RE',
    'S1',
    'L2',
    'G2'
)
AND billing_document_number IS NOT NULL

GROUP BY
    billing_document_number,
    sales_order_item,
    material_number

HAVING COUNT(additional_quantity_in_tons) > 1

LIMIT 100;
