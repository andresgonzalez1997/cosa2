WITH shipping_base AS (

    SELECT
        order_number,
        order_line_number,
        product_id,
        SUM(quantity_ordered_item_unique_in_tons) AS toneladas_embarcadas

    FROM prd_product_anh_external_datablocks.shipping

    WHERE quantity_ordered_item_unique_in_tons IS NOT NULL

    GROUP BY
        order_number,
        order_line_number,
        product_id
),


org_base AS (

    SELECT
        org_id,
        MAX(country_name) AS country_name

    FROM prd_product_anh_core_data.org_location_core

    GROUP BY
        org_id
),


customer_base AS (

    SELECT
        customer_shipto_id,
        MAX(customer_shipto_country_code) AS customer_shipto_country_code,
        MAX(customer_soldto_price_list_erp) AS customer_soldto_price_list_erp,
        MAX(customer_shipto_segment_code_erp) AS customer_shipto_segment_code_erp,
        MAX(customer_segment_name) AS customer_segment_name,
        MAX(customer_soldto_number) AS customer_central_number

    FROM prd_product_anh_core_data.customer_core

    GROUP BY
        customer_shipto_id
),


product_base AS (

    SELECT
        product_id,
        MAX(species_l2) AS species_l2,
        MAX(sub_species_l3) AS sub_species_l3,
        MAX(form_code) AS form_code,
        MAX(container) AS container,
        MAX(unit_type_code) AS unit_type_code

    FROM prd_product_anh_core_data.product_core

    GROUP BY
        product_id
),


net_contents_base AS (

    SELECT
        material_number,
        MAX(characteristic_value) AS net_contents

    FROM prd_product_anh_internal_datablocks.tca_material_characteristics_assignments

    WHERE characteristic_description = 'NET CONTENTS'

    GROUP BY
        material_number
),


profit_base AS (

    SELECT
        p.*,

        MAX(p.additional_quantity_in_tons) OVER (
            PARTITION BY
                p.sales_order_number,
                p.sales_order_item,
                p.product_id,
                p.material_number,
                p.org_id,
                p.customer_shipto_id
        ) AS toneladas_facturadas_linea,

        ROW_NUMBER() OVER (
            PARTITION BY
                p.sales_order_number,
                p.sales_order_item,
                p.product_id,
                p.material_number,
                p.org_id,
                p.customer_shipto_id

            ORDER BY
                p.accounting_document_number DESC,
                p.ledger_posting_item DESC
        ) AS rn_line

    FROM prd_product_anh_external_datablocks.tca_enhanced_profit_and_loss p
),


main_data AS (

    SELECT

        org.country_name AS Ship_From,

        profit.sales_org AS Sales_Org,

        customer.customer_shipto_country_code AS Ship_To,

        customer.customer_segment_name AS Customer_Segmentation,

        customer.customer_shipto_segment_code_erp AS customer_segment,


        COALESCE(
            MIN(
                CASE
                    WHEN profit.cfmr_account_level2 = 'NET SALES'
                    THEN profit.posting_date
                END
            ),
            MIN(profit.posting_date)
        ) AS Date_,


        profit.sales_order_number AS Order_Number,

        profit.sales_order_item AS Order_Line,

        customer.customer_central_number AS customer_id,

        profit.customer_sold_to_number AS customer_sold_to_number,

        profit.source_customer_sold_to_name AS Customer_Name,

        product.species_l2 AS Product_Category,

        product.sub_species_l3 AS Product_Sub_Specie,

        profit.material_number AS Product_ID,

        profit.sap_material_description AS Product_Name,


        COALESCE(
            MIN(
                CASE
                    WHEN profit.cfmr_account_level2 = 'NET SALES'
                    THEN profit.fiscal_year
                END
            ),
            MIN(profit.fiscal_year)
        ) AS FY,


        COALESCE(
            MIN(
                CASE
                    WHEN profit.cfmr_account_level2 = 'NET SALES'
                    THEN profit.posting_period
                END
            ),
            MIN(profit.posting_period)
        ) AS Month_,


        customer.customer_soldto_price_list_erp AS Price_List,


        COALESCE(
            MAX(profit.sales_representative_name),
            'SIN_VENDEDOR'
        ) AS Sales_Person,


        product.form_code AS Form,

        product.container AS Container,

        product.unit_type_code AS empaque,

        net_contents_base.net_contents AS Net_Contents,


        COALESCE(
            SUM(
                CASE
                    WHEN profit.rn_line = 1
                    THEN profit.toneladas_facturadas_linea
                    ELSE 0
                END
            ),
            0
        ) AS Metric_Tons,


        CAST(
            SUM(
                CASE
                    WHEN profit.cfmr_account_level1 = 'GROSS SALES'
                     AND UPPER(profit.cfmr_account_description) NOT LIKE '%FREIGHT%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Gross_Sales_USD,


        CAST(
            SUM(
                CASE
                    WHEN profit.cfmr_account_level1 = 'GROSS SALES'
                     AND UPPER(profit.cfmr_account_description) LIKE '%FREIGHT%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Freight_Revenue_USD,


        CAST(
            SUM(
                CASE
                    WHEN profit.cfmr_account_level1 = 'GROSS SALES'
                    THEN profit.amount_company_code_currency
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Gross_Sales_Local,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_description) LIKE '%DISCOUNT%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Discount_USD,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_description) LIKE '%DISCOUNT%'
                    THEN profit.amount_company_code_currency
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Discount_Local,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_level1) LIKE '%DEPRECIATION%'
                      OR UPPER(profit.cfmr_account_level1) LIKE '%DEPR EXP%'
                      OR UPPER(profit.cfmr_account_level1) LIKE '%AMORT EXP%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS COGS_Depreciation_USD,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_level1) LIKE '%MANUFACTURING%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS COGS_Manufacturing_USD,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_level1) LIKE '%PACKAGING%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS COGS_Packaging_USD,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_level1) LIKE '%INGREDIENT%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS COGS_Ingredients_USD,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_level1) LIKE '%COGS%'

                     AND UPPER(profit.cfmr_account_level1)
                         NOT LIKE '%DEPRECIATION%'

                     AND UPPER(profit.cfmr_account_level1)
                         NOT LIKE '%DEPR EXP%'

                     AND UPPER(profit.cfmr_account_level1)
                         NOT LIKE '%AMORT EXP%'

                     AND UPPER(profit.cfmr_account_level1)
                         NOT LIKE '%MANUFACTURING%'

                     AND UPPER(profit.cfmr_account_level1)
                         NOT LIKE '%PACKAGING%'

                     AND UPPER(profit.cfmr_account_level1)
                         NOT LIKE '%INGREDIENT%'

                    THEN profit.amount_usd

                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS COGS_Otros_USD,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_level1) LIKE '%COGS%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS COGS_Total_USD,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_level1) LIKE '%COGS%'
                    THEN profit.amount_company_code_currency
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS COGS_Total_Local,


        /* ====================================================
           CAMBIO:
           Material Cost LATAM según jerarquía CFMR real
           ==================================================== */
        CAST(
            SUM(
                CASE
                    WHEN UPPER(TRIM(profit.cfmr_account_level2))
                            = 'COST OF INVENTORY SOLD'
                     AND UPPER(TRIM(profit.cfmr_account_level3))
                            = 'MATERIAL COSTS'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Material_Cost_LATAM_USD,


        CAST(

            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_level1) LIKE '%MANUFACTURING%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )

            +

            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_level1) LIKE '%DEPRECIATION%'
                      OR UPPER(profit.cfmr_account_level1) LIKE '%DEPR EXP%'
                      OR UPPER(profit.cfmr_account_level1) LIKE '%AMORT EXP%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )

            +

            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_level1) LIKE '%COGS%'

                     AND UPPER(profit.cfmr_account_level1)
                         NOT LIKE '%DEPRECIATION%'

                     AND UPPER(profit.cfmr_account_level1)
                         NOT LIKE '%DEPR EXP%'

                     AND UPPER(profit.cfmr_account_level1)
                         NOT LIKE '%AMORT EXP%'

                     AND UPPER(profit.cfmr_account_level1)
                         NOT LIKE '%MANUFACTURING%'

                     AND UPPER(profit.cfmr_account_level1)
                         NOT LIKE '%PACKAGING%'

                     AND UPPER(profit.cfmr_account_level1)
                         NOT LIKE '%INGREDIENT%'

                    THEN profit.amount_usd

                    ELSE 0
                END
            )

            +

            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_description) LIKE '%FREIGHT%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )

            AS DECIMAL(18,3)

        ) AS Other_Costs_LATAM_USD,


        CAST(
            SUM(
                CASE
                    WHEN profit.cfmr_account_level5 = 'MATERIAL MARGIN'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Material_Margin_USD,


        CAST(
            SUM(
                CASE
                    WHEN profit.cfmr_account_level5 = 'MATERIAL MARGIN'
                    THEN profit.amount_company_code_currency
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Material_Margin_Local,


        CAST(
            SUM(
                CASE
                    WHEN profit.cfmr_account_level2 = 'COST OF INVENTORY SOLD'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Cost_Inv_Sold_USD,


        CAST(
            SUM(
                CASE
                    WHEN profit.cfmr_account_level2 = 'COST OF INVENTORY SOLD'
                    THEN profit.amount_company_code_currency
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Cost_Inv_Sold_Local,


        CAST(
            SUM(
                CASE
                    WHEN profit.cfmr_account_level2 = 'NET SALES'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Net_Sales_USD,


        CAST(
            SUM(
                CASE
                    WHEN profit.cfmr_account_level2 = 'NET SALES'
                    THEN profit.amount_company_code_currency
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Net_Sales_Local,


        CAST(
            SUM(
                CASE
                    WHEN profit.cfmr_account_level2 = 'LOGISTICS'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Logistics_USD,


        CAST(
            SUM(
                CASE
                    WHEN profit.cfmr_account_level2 = 'LOGISTICS'
                    THEN profit.amount_company_code_currency
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Logistics_Local,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.billing_type_code) LIKE '%G2%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Credit_Memo_USD,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.billing_type_code) LIKE '%G2%'
                    THEN profit.amount_company_code_currency
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Credit_Memo_Local,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_description) LIKE '%FREIGHT%'
                    THEN profit.amount_usd
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Freight_USD,


        CAST(
            SUM(
                CASE
                    WHEN UPPER(profit.cfmr_account_description) LIKE '%FREIGHT%'
                    THEN profit.amount_company_code_currency
                    ELSE 0
                END
            )
            AS DECIMAL(18,3)
        ) AS Freight_Local,


        shipping_base.order_number AS order_number_shipping,


        COALESCE(
            SUM(
                CASE
                    WHEN profit.rn_line = 1
                    THEN COALESCE(profit.toneladas_facturadas_linea, 0)
                    ELSE 0
                END
            ),
            0
        ) AS toneladas_facturadas,


        COALESCE(
            SUM(
                CASE
                    WHEN profit.rn_line = 1
                    THEN COALESCE(shipping_base.toneladas_embarcadas, 0)
                    ELSE 0
                END
            ),
            0
        ) AS toneladas_embarcadas,


        CASE
            WHEN MIN(profit.fiscal_year) <> MAX(profit.fiscal_year)
              OR MIN(profit.posting_period) <> MAX(profit.posting_period)
            THEN 'SI'

            ELSE 'NO'

        END AS Costeo_Ajustado_Periodo_Posterior


    FROM profit_base AS profit


    LEFT JOIN org_base AS org
        ON profit.org_id = org.org_id


    LEFT JOIN customer_base AS customer
        ON profit.customer_shipto_id = customer.customer_shipto_id


    LEFT JOIN product_base AS product
        ON profit.product_id = product.product_id


    LEFT JOIN net_contents_base
        ON profit.material_number = net_contents_base.material_number


    LEFT JOIN shipping_base
        ON profit.sales_order_number = shipping_base.order_number

       AND profit.sales_order_item = shipping_base.order_line_number

       AND (
            profit.product_id = shipping_base.product_id
            OR (
                profit.product_id IS NULL
                AND shipping_base.product_id IS NULL
            )
       )


    WHERE org.country_name IN (
        'Guatemala',
        'Honduras',
        'Nicaragua',
        'El Salvador',
        'Costa Rica',
        'Panama',
        'Aruba',
        'Republica Dominicana',
        'Puerto Rico'
    )


    AND (
        customer.customer_central_number IS NULL

        OR customer.customer_central_number NOT IN (
            '1000180708',
            '1000068657',
            '1000023782',
            '1800098231',
            '1000180648',
            '1800098230',
            '1000180647'
        )
    )


    AND (
        profit.billing_type_code IS NULL
        OR UPPER(profit.billing_type_code) NOT LIKE 'IV%'
    )


    AND profit.material_type = 'ZMAT'


    GROUP BY

        org.country_name,

        profit.sales_org,

        customer.customer_shipto_country_code,

        customer.customer_soldto_price_list_erp,

        product.species_l2,

        product.sub_species_l3,

        customer.customer_shipto_segment_code_erp,

        customer.customer_segment_name,

        profit.sap_material_description,

        product.form_code,

        product.container,

        product.unit_type_code,

        net_contents_base.net_contents,

        profit.source_customer_sold_to_name,

        customer.customer_central_number,

        profit.material_number,

        profit.customer_sold_to_number,

        profit.sales_order_number,

        profit.sales_order_item,

        shipping_base.order_number


    HAVING

        COALESCE(
            SUM(
                CASE
                    WHEN profit.rn_line = 1
                    THEN profit.toneladas_facturadas_linea
                    ELSE 0
                END
            ),
            0
        ) > 0

        AND (

            CAST(
                SUM(
                    CASE
                        WHEN profit.cfmr_account_level2 = 'NET SALES'
                        THEN profit.amount_usd
                        ELSE 0
                    END
                )
                AS DECIMAL(18,3)
            ) <> 0

            OR

            CAST(
                SUM(
                    CASE
                        WHEN UPPER(profit.billing_type_code) LIKE '%G2%'
                        THEN profit.amount_usd
                        ELSE 0
                    END
                )
                AS DECIMAL(18,3)
            ) <> 0
        )
),


/* ============================================================
   COSTO HN10
   ============================================================ */

hn10_cost_line AS (

    SELECT

        profit.material_number AS Product_ID,

        profit.fiscal_year AS FY,

        profit.posting_period AS Month_,

        profit.posting_date AS Cost_Posting_Date,

        profit.sales_order_number,

        profit.sales_order_item,


        MAX(
            COALESCE(
                profit.toneladas_facturadas_linea,
                0
            )
        ) AS Cost_Tons,


        /* ====================================================
           CAMBIO:
           Material Cost según CFMR real
           ==================================================== */
        SUM(
            CASE
                WHEN UPPER(TRIM(profit.cfmr_account_level2))
                        = 'COST OF INVENTORY SOLD'
                 AND UPPER(TRIM(profit.cfmr_account_level3))
                        = 'MATERIAL COSTS'
                THEN profit.amount_usd
                ELSE 0
            END
        ) AS Material_Cost_USD


    FROM profit_base profit


    WHERE UPPER(profit.sales_org) = 'HN10'

      AND profit.material_type = 'ZMAT'


    GROUP BY

        profit.material_number,

        profit.fiscal_year,

        profit.posting_period,

        profit.posting_date,

        profit.sales_order_number,

        profit.sales_order_item


    HAVING

        MAX(
            COALESCE(
                profit.toneladas_facturadas_linea,
                0
            )
        ) > 0

        AND

        /* ====================================================
           CAMBIO:
           validación del costo usando jerarquía CFMR real
           ==================================================== */
        SUM(
            CASE
                WHEN UPPER(TRIM(profit.cfmr_account_level2))
                        = 'COST OF INVENTORY SOLD'
                 AND UPPER(TRIM(profit.cfmr_account_level3))
                        = 'MATERIAL COSTS'
                THEN profit.amount_usd
                ELSE 0
            END
        ) <> 0
),


hn10_cost_daily AS (

    SELECT

        Product_ID,

        FY,

        Month_,

        Cost_Posting_Date,


        SUM(Material_Cost_USD) AS Material_Cost_USD,

        SUM(Cost_Tons) AS Cost_Tons,


        CASE

            WHEN SUM(Cost_Tons) <> 0

            THEN
                ABS(SUM(Material_Cost_USD))
                /
                SUM(Cost_Tons)

            ELSE NULL

        END AS hn10_material_cost_usd_per_ton


    FROM hn10_cost_line


    GROUP BY

        Product_ID,

        FY,

        Month_,

        Cost_Posting_Date
),


hn10_cost_ranked AS (

    SELECT

        Product_ID,

        FY,

        Month_,

        Cost_Posting_Date,

        hn10_material_cost_usd_per_ton,


        ROW_NUMBER() OVER (

            PARTITION BY

                Product_ID,

                FY,

                Month_


            ORDER BY

                Cost_Posting_Date DESC

        ) AS rn_cost


    FROM hn10_cost_daily


    WHERE hn10_material_cost_usd_per_ton IS NOT NULL
),


hn10_cost_base AS (

    SELECT

        Product_ID,

        FY,

        Month_,

        Cost_Posting_Date AS Latest_Cost_Date,

        hn10_material_cost_usd_per_ton


    FROM hn10_cost_ranked


    WHERE rn_cost = 1
),


classified_data AS (

    SELECT

        main_data.*,

        hn.Latest_Cost_Date,

        hn.hn10_material_cost_usd_per_ton,


        CONCAT(
            main_data.Ship_From,
            ' -> ',
            main_data.Ship_To
        ) AS Route_Key,


        CASE

            WHEN main_data.Product_ID IN (
                '4053_AM_PL_20KG',
                '4053_PT_PL_20KG',
                '4054_PT_PL_25KG',
                '4056_LE_PL_25KG'
            )
            THEN 'Specific Pork Products'


            WHEN UPPER(main_data.Product_Category) LIKE '%PET%'
            THEN 'Pet Food'


            WHEN UPPER(main_data.Product_Category) LIKE '%TILAPIA%'
            THEN 'Tilapia'


            ELSE 'Feed Other'

        END AS Process_Family,


        CASE

            WHEN UPPER(main_data.Sales_Org) = 'HN10'
            THEN 'Normal'


            WHEN

                main_data.Product_ID NOT IN (
                    '4053_AM_PL_20KG',
                    '4053_PT_PL_20KG',
                    '4054_PT_PL_25KG',
                    '4056_LE_PL_25KG'
                )

                AND UPPER(main_data.Product_Category) NOT LIKE '%PET%'

                AND UPPER(main_data.Product_Category) NOT LIKE '%TILAPIA%'

            THEN

                CASE

                    WHEN UPPER(main_data.Sales_Org) = 'HN20'
                    THEN 'E2E'

                    ELSE 'Normal'

                END


            WHEN UPPER(main_data.Sales_Org) IN (
                'HN20',
                'GT10',
                'NI10',
                'CR10'
            )
            THEN 'E2E'


            ELSE 'Normal'

        END AS E2E_Flag


    FROM main_data


    LEFT JOIN hn10_cost_base hn

        ON main_data.Product_ID = hn.Product_ID

       AND main_data.FY = hn.FY

       AND main_data.Month_ = hn.Month_
)


SELECT

    *,


    CASE

        WHEN E2E_Flag = 'Normal'
        THEN 'NORMAL'

        WHEN hn10_material_cost_usd_per_ton IS NULL
        THEN 'MISSING_HN10_COST_SAME_MONTH'

        ELSE 'OK'

    END AS E2E_Cost_Status,


    CASE

        WHEN E2E_Flag = 'E2E'
         AND hn10_material_cost_usd_per_ton IS NOT NULL

        THEN
            hn10_material_cost_usd_per_ton
            * toneladas_facturadas


        WHEN E2E_Flag = 'E2E'
        THEN NULL


        ELSE 0

    END AS Material_Cost_LATAM_E2E_USD,


    CASE

        WHEN E2E_Flag = 'E2E'
        THEN ABS(Logistics_USD)

        ELSE 0

    END AS Logistics_E2E_USD,


    CASE

        WHEN E2E_Flag = 'E2E'
         AND hn10_material_cost_usd_per_ton IS NOT NULL

        THEN

              Gross_Sales_USD

            - ABS(Discount_USD)

            - (
                hn10_material_cost_usd_per_ton
                * toneladas_facturadas
              )

            - ABS(Logistics_USD)


        WHEN E2E_Flag = 'Normal'
        THEN Material_Margin_USD


        ELSE NULL

    END AS Margin_E2E_USD,


    CASE

        WHEN E2E_Flag = 'E2E'
         AND hn10_material_cost_usd_per_ton IS NOT NULL

        THEN
            hn10_material_cost_usd_per_ton
            * toneladas_facturadas


        WHEN E2E_Flag = 'E2E'
        THEN NULL


        ELSE 0

    END AS Cost_E2E_USD,


    CASE

        WHEN E2E_Flag = 'E2E'
        THEN toneladas_facturadas

        ELSE 0

    END AS TM_Facturado_E2E,


    CASE

        WHEN E2E_Flag = 'E2E'
        THEN toneladas_embarcadas

        ELSE 0

    END AS TM_Embarcado_E2E


FROM classified_data;
