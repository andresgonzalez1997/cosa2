A. org_location_core

SELECT
    org_id,
    COUNT(*) AS qty
FROM prd_product_anh_core_data.org_location_core
WHERE org_id IS NOT NULL
GROUP BY org_id
HAVING COUNT(*) > 1
ORDER BY qty DESC
LIMIT 100;




B. customer_core

SELECT
    customer_shipto_id,
    COUNT(*) AS qty
FROM prd_product_anh_core_data.customer_core
WHERE customer_shipto_id IS NOT NULL
GROUP BY customer_shipto_id
HAVING COUNT(*) > 1
ORDER BY qty DESC
LIMIT 100;




C. product_core

SELECT
    product_id,
    COUNT(*) AS qty
FROM prd_product_anh_core_data.product_core
WHERE product_id IS NOT NULL
GROUP BY product_id
HAVING COUNT(*) > 1
ORDER BY qty DESC
LIMIT 100;
