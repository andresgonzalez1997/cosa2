WITH shipping_base AS (
    SELECT
        order_number,
        order_line_number,
        product_id,
        SUM(quantity_shipped_and_invoiced_in_tons) AS toneladas_embarcadas
    FROM prd_product_anh_external_datablocks.shipping
    GROUP BY
        order_number,
        order_line_number,
        product_id
),

org_base AS (
    SELECT
        org_id,
        MAX(country_name) AS country_name
    FROM prd_product_anh_core_data.org_location_core
    GROUP BY
        org_id
),

customer_base AS (
    SELECT
        customer_shipto_id,
        MAX(customer_shipto_country_code) AS customer_shipto_country_code,
        MAX(customer_soldto_price_list_erp) AS customer_soldto_price_list_erp,
        MAX(customer_shipto_segment_code_erp) AS customer_shipto_segment_code_erp,
        MAX(customer_segment_name) AS customer_segment_name,
        MAX(customer_central_number) AS customer_central_number
    FROM prd_product_anh_core_data.customer_core
    GROUP BY
        customer_shipto_id
),

product_base AS (
    SELECT
        product_id,
        MAX(species_12) AS species_12,
        MAX(sub_species_13) AS sub_species_13,
        MAX(form_code) AS form_code,
        MAX(container) AS container,
        MAX(unit_type_code) AS unit_type_code
    FROM prd_product_anh_core_data.product_core
    GROUP BY
        product_id
),

profit_base AS (
    SELECT
        p.*,

        MAX(p.additional_quantity_in_tons) OVER (
            PARTITION BY
                p.sales_order_number,
                p.sales_order_item,
                p.product_id,
                p.material_number,
                p.org_id,
                p.customer_shipto_id,
                p.fiscal_year,
                p.posting_period
        ) AS toneladas_facturadas_linea,

        ROW_NUMBER() OVER (
            PARTITION BY
                p.sales_order_number,
                p.sales_order_item,
                p.product_id,
                p.material_number,
                p.org_id,
                p.customer_shipto_id,
                p.fiscal_year,
                p.posting_period
            ORDER BY
                p.posting_date DESC,
                p.accounting_document_number DESC,
                p.ledger_posting_item DESC
        ) AS rn_line

    FROM prd_product_anh_external_datablocks.tca_enhanced_profit_and_loss p
)

SELECT
    /*
    CONCAT_WS('-',
        profit.company_code,
        profit.accounting_document_number,
        profit.ledger_posting_item,
        CAST(profit.product_id AS STRING)
    ) AS id_doc,
    */

    profit.fiscal_year AS FY,
    profit.posting_period AS Month_,
    org.country_name AS Ship_From,
    customer.customer_shipto_country_code AS Ship_To,
    customer.customer_soldto_price_list_erp AS Price_List,
    product.species_12 AS Product_Category,
    product.sub_species_13 AS Product_Sub_Specie,
    customer.customer_shipto_segment_code_erp AS customer_segment,
    customer.customer_segment_name AS Customer_Segmentation,
    COALESCE(MAX(profit.sales_representative_name), 'SIN_VENDEDOR') AS Sales_Person,
    profit.sap_material_description AS Product_Name,
    product.form_code AS Form,
    product.container AS Container,
    product.unit_type_code AS empaque,
    profit.source_customer_sold_to_name AS Customer_Name,
    customer.customer_central_number AS customer_id,

    COALESCE(
        SUM(
            CASE
                WHEN profit.rn_line = 1 THEN profit.toneladas_facturadas_linea
                ELSE 0
            END
        ),
        0
    ) AS Metric_Tons,

    CAST(
        SUM(
            CASE
                WHEN profit.cfmr_account_level1 = 'GROSS SALES'
                THEN profit.amount_usd
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Gross_Sales_USD,

    CAST(
        SUM(
            CASE
                WHEN profit.cfmr_account_level1 = 'GROSS SALES'
                THEN profit.amount_company_code_currency
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Gross_Sales_Local,

    CAST(
        SUM(
            CASE
                WHEN profit.cfmr_account_level2 = 'COST OF INVENTORY SOLD'
                THEN profit.amount_usd
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Cost_Inv_Sold_USD,

    CAST(
        SUM(
            CASE
                WHEN profit.cfmr_account_level2 = 'COST OF INVENTORY SOLD'
                THEN profit.amount_company_code_currency
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Cost_Inv_Sold_Local,

    CAST(
        SUM(
            CASE
                WHEN UPPER(profit.cfmr_account_description) LIKE '%DISCOUNT%'
                THEN profit.amount_usd
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Discount_USD,

    CAST(
        SUM(
            CASE
                WHEN UPPER(profit.cfmr_account_description) LIKE '%DISCOUNT%'
                THEN profit.amount_company_code_currency
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Discount_Local,

    CAST(
        SUM(
            CASE
                WHEN profit.cfmr_account_level5 = 'MATERIAL MARGIN'
                THEN profit.amount_usd
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Material_Margin_USD,

    CAST(
        SUM(
            CASE
                WHEN profit.cfmr_account_level5 = 'MATERIAL MARGIN'
                THEN profit.amount_company_code_currency
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Material_Margin_Local,

    CAST(
        SUM(
            CASE
                WHEN profit.cfmr_account_level2 = 'NET SALES'
                THEN profit.amount_usd
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Net_Sales_USD,

    CAST(
        SUM(
            CASE
                WHEN profit.cfmr_account_level2 = 'NET SALES'
                THEN profit.amount_company_code_currency
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Net_Sales_Local,

    CAST(
        SUM(
            CASE
                WHEN UPPER(profit.cfmr_account_level1) LIKE '%COGS%'
                THEN profit.amount_usd
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS COGS_USD,

    CAST(
        SUM(
            CASE
                WHEN UPPER(profit.cfmr_account_level1) LIKE '%COGS%'
                THEN profit.amount_company_code_currency
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS COGS_Local,

    CAST(
        SUM(
            CASE
                WHEN profit.cfmr_account_level2 = 'LOGISTICS'
                THEN profit.amount_usd
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Logistics_USD,

    CAST(
        SUM(
            CASE
                WHEN profit.cfmr_account_level2 = 'LOGISTICS'
                THEN profit.amount_company_code_currency
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Logistics_Local,

    CAST(
        SUM(
            CASE
                WHEN UPPER(profit.billing_type_code) LIKE '%G2%'
                THEN profit.amount_usd
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Credit_Memo_USD,

    CAST(
        SUM(
            CASE
                WHEN UPPER(profit.billing_type_code) LIKE '%G2%'
                THEN profit.amount_company_code_currency
                ELSE 0
            END
        ) AS DECIMAL(18,3)
    ) AS Credit_Memo_Local,

    profit.material_number AS Product_ID,
    MAX(profit.posting_date) AS Date_,
    shipping_base.order_number AS order_number_shipping,

    COALESCE(
        SUM(
            CASE
                WHEN profit.rn_line = 1 THEN profit.toneladas_facturadas_linea
                ELSE 0
            END
        ),
        0
    ) AS toneladas_facturadas,

    COALESCE(
        SUM(
            CASE
                WHEN profit.rn_line = 1 THEN COALESCE(shipping_base.toneladas_embarcadas, 0)
                ELSE 0
            END
        ),
        0
    ) AS toneladas_embarcadas

FROM profit_base AS profit

LEFT JOIN org_base AS org
    ON profit.org_id = org.org_id

LEFT JOIN customer_base AS customer
    ON profit.customer_shipto_id = customer.customer_shipto_id

LEFT JOIN product_base AS product
    ON profit.product_id = product.product_id

LEFT JOIN shipping_base
    ON profit.sales_order_number = shipping_base.order_number
   AND profit.sales_order_item = shipping_base.order_line_number
   AND profit.product_id = shipping_base.product_id

WHERE org.country_name IN ('Mexico')
  AND profit.sales_order_number = '1002089809'
--AND profit.material_number = '0000000000130131481'

GROUP BY
    profit.fiscal_year,
    profit.posting_period,
    org.country_name,
    customer.customer_shipto_country_code,
    customer.customer_soldto_price_list_erp,
    product.species_12,
    product.sub_species_13,
    customer.customer_shipto_segment_code_erp,
    customer.customer_segment_name,
    profit.sap_material_description,
    product.form_code,
    product.container,
    product.unit_type_code,
    profit.source_customer_sold_to_name,
    customer.customer_central_number,
    profit.material_number,
    shipping_base.order_number
;
