SELECT
    COUNT(*) AS rows_profit,

    SUM(
        CASE
            WHEN org.org_id IS NULL
            THEN 1
            ELSE 0
        END
    ) AS rows_without_org,

    SUM(
        CASE
            WHEN customer.customer_shipto_id IS NULL
            THEN 1
            ELSE 0
        END
    ) AS rows_without_customer,

    SUM(
        CASE
            WHEN product.product_id IS NULL
            THEN 1
            ELSE 0
        END
    ) AS rows_without_product

FROM prd_product_anh_external_datablocks.tca_enhanced_profit_and_loss profit

LEFT JOIN prd_product_anh_core_data.org_location_core org
    ON profit.org_id = org.org_id

LEFT JOIN prd_product_anh_core_data.customer_core customer
    ON profit.customer_shipto_id = customer.customer_shipto_id

LEFT JOIN prd_product_anh_core_data.product_core product
    ON profit.product_id = product.product_id

WHERE profit.billing_type_code IN (
    'ZUFD',
    'RE',
    'S1',
    'L2',
    'G2'
);
